cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(InputCIRunner VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Use Visual Studio 2022 C++ tools for Windows x64.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(SDK_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/SDK" CACHE PATH "Vdevice SDK directory; override privately for the production edition")
option(ENABLE_ASAN "Enable AddressSanitizer for offline tests" OFF)
add_compile_definitions(UNICODE _UNICODE NOMINMAX)
add_compile_options(/W4 /WX /utf-8 /permissive- /EHsc /guard:cf)
add_link_options(/guard:cf /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /PDBALTPATH:%_PDB%)
if(ENABLE_ASAN)
 set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
 add_compile_options(/fsanitize=address /Zi)
 add_link_options(/DEBUG /INCREMENTAL:NO)
endif()
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
 set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
endforeach()
add_library(RunnerCore STATIC App/Fixture.cpp App/Runner.cpp "${SDK_DIRECTORY}/Vdevice.cpp")
target_include_directories(RunnerCore PUBLIC App "${SDK_DIRECTORY}")
add_executable(InputCIRunner App/Main.cpp)
target_link_libraries(InputCIRunner PRIVATE RunnerCore)
target_link_options(InputCIRunner PRIVATE "/MANIFESTUAC:level='asInvoker' uiAccess='false'")
add_executable(RunnerTests Tests/RunnerTests.cpp)
target_link_libraries(RunnerTests PRIVATE RunnerCore)
enable_testing()
add_test(NAME OfflineModels COMMAND RunnerTests)
add_test(NAME OfflineApplication COMMAND InputCIRunner --self-test)
add_test(NAME OfflineMouse COMMAND InputCIRunner validate "${CMAKE_CURRENT_SOURCE_DIR}/Examples/mouse.json")
add_test(NAME OfflineKeyboard COMMAND InputCIRunner validate "${CMAKE_CURRENT_SOURCE_DIR}/Examples/keyboard.json")
